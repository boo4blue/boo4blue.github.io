One file

An entire game in one .Z file

Scenes, scripts, UI, and web — one format. Write it, and Z Engine runs it on any phone.

# mygame.Z — a whole game in one file
scene "MyGame" {
    sky: sunset
    ground: desert
    prop hero {
        mesh: ball
        at: [0, 0.9, 0]
        body: sphere
    }
}

script {
    let hero = 0
    fn on_start() {
        hero = find("hero")
        play_sound("game_start")
    }
    fn on_tap(x, y, z) {
        let p = get_pos(hero)
        push(hero, (x - p[0]) * 3, 7, (z - p[2]) * 3)
        burst(p[0], p[1], p[2], 1, 0.8, 0.3, 20, 8)
    }
}
